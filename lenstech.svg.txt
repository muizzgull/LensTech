    <!-- Three.js 3D Dismantling Keyboard Engine -->
    <script>
        // --- Mobile Menu Toggle ---
        const menuBtn = document.getElementById('mobile-menu-btn');
        const mobileMenu = document.getElementById('mobile-menu');
        menuBtn.addEventListener('click', () => {
            mobileMenu.classList.toggle('hidden');
            mobileMenu.classList.toggle('flex');
        });

        // --- Three.js Setup ---
        const container = document.getElementById('hero-3d-container');
        const scene = new THREE.Scene();

        // Camera Setup
        const camera = new THREE.PerspectiveCamera(50, window.innerWidth / window.innerHeight, 0.1, 1000);
        camera.position.set(0, 22, 38);
        camera.lookAt(0, 0, 0);

        // Renderer
        const renderer = new THREE.WebGLRenderer({ alpha: true, antialias: true });
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
        container.appendChild(renderer.domElement);

        // --- Lighting ---
        const ambientLight = new THREE.AmbientLight(0xffffff, 1.4);
        scene.add(ambientLight);

        const mainLight = new THREE.DirectionalLight(0xffffff, 1.8);
        mainLight.position.set(20, 40, 20);
        scene.add(mainLight);

        const accentLight = new THREE.PointLight(0x6366f1, 2.5, 100);
        accentLight.position.set(-20, -10, 10);
        scene.add(accentLight);

        // --- 3D Keyboard Assembly Groups ---
        const keyboardGroup = new THREE.Group();
        keyboardGroup.rotation.x = 0.55; // Tilt keyboard upward toward user
        keyboardGroup.rotation.y = -0.3;

        // Layer 1: Base Case (Bottom)
        const caseGeo = new THREE.BoxGeometry(26, 1.5, 11);
        const caseMat = new THREE.MeshStandardMaterial({
            color: 0x1e293b, // Dark slate metal
            roughness: 0.3,
            metalness: 0.8
        });
        const caseMesh = new THREE.Mesh(caseGeo, caseMat);
        caseMesh.position.y = -1;
        keyboardGroup.add(caseMesh);

        // Layer 2: Circuit Board / Plate (Middle)
        const pcbGeo = new THREE.BoxGeometry(25, 0.4, 10);
        const pcbMat = new THREE.MeshStandardMaterial({
            color: 0x4f46e5, // Brand Indigo PCB
            roughness: 0.2,
            metalness: 0.5
        });
        const pcbMesh = new THREE.Mesh(pcbGeo, pcbMat);
        pcbMesh.position.y = 0;
        keyboardGroup.add(pcbMesh);

        // Layer 3: Key Switches Group
        const switchesGroup = new THREE.Group();
        const switchGeo = new THREE.BoxGeometry(0.9, 0.6, 0.9);
        const switchMat = new THREE.MeshStandardMaterial({
            color: 0x06b6d4, // Cyan mechanical switches
            metalness: 0.3,
            roughness: 0.2
        });

        // Layer 4: Keycaps Group
        const keycapsGroup = new THREE.Group();
        const keycapGeo = new THREE.BoxGeometry(1.1, 0.8, 1.1);
        const keycapMat = new THREE.MeshStandardMaterial({
            color: 0xf8fafc, // Crisp off-white keycaps
            roughness: 0.4,
            metalness: 0.1
        });
        const accentKeycapMat = new THREE.MeshStandardMaterial({
            color: 0x6366f1, // Accent color for special keys
            roughness: 0.3,
            metalness: 0.2
        });

        // Generate Keyboard Grid (14 columns x 5 rows)
        const cols = 14;
        const rows = 5;
        const spacingX = 1.6;
        const spacingZ = 1.6;
        const startX = -((cols - 1) * spacingX) / 2;
        const startZ = -((rows - 1) * spacingZ) / 2;

        const keycapsList = [];
        const switchesList = [];

        for (let r = 0; r < rows; r++) {
            for (let c = 0; c < cols; c++) {
                const posX = startX + c * spacingX;
                const posZ = startZ + r * spacingZ;

                // Create Switch
                const switchMesh = new THREE.Mesh(switchGeo, switchMat);
                switchMesh.position.set(posX, 0.6, posZ);
                switchesGroup.add(switchMesh);
                switchesList.push(switchMesh);

                // Create Keycap (use accent color for outer keys)
                const isAccent = (c === 0 || c === cols - 1 || r === 0 || r === rows - 1);
                const capMat = isAccent ? accentKeycapMat : keycapMat;
                const keycapMesh = new THREE.Mesh(keycapGeo, capMat);
                keycapMesh.position.set(posX, 1.4, posZ);
                
                // Store initial relative offsets for staggered explosion effect
                keycapMesh.userData = {
                    initialY: 1.4,
                    delayFactor: (Math.abs(c - cols/2) + Math.abs(r - rows/2)) * 0.15
                };

                keycapsGroup.add(keycapMesh);
                keycapsList.push(keycapMesh);
            }
        }

        keyboardGroup.add(switchesGroup);
        keyboardGroup.add(keycapsGroup);

        // Floating Ambient Light Particles around keyboard
        const particleCount = 200;
        const particleGeo = new THREE.BufferGeometry();
        const positions = new Float32Array(particleCount * 3);

        for (let i = 0; i < particleCount * 3; i += 3) {
            positions[i] = (Math.random() - 0.5) * 60;
            positions[i + 1] = (Math.random() - 0.5) * 40;
            positions[i + 2] = (Math.random() - 0.5) * 40;
        }

        particleGeo.setAttribute('position', new THREE.BufferAttribute(positions, 3));
        const particleMat = new THREE.PointsMaterial({
            color: 0x818cf8,
            size: 0.35,
            transparent: true,
            opacity: 0.6
        });
        const particles = new THREE.Points(particleGeo, particleMat);
        keyboardGroup.add(particles);

        scene.add(keyboardGroup);

        // --- Pinned Scroll Logic ---
        const pinWrapper = document.getElementById('hero-pin-wrapper');
        const heroContent = document.getElementById('hero-content');

        let scrollProgress = 0;

        function updateScrollProgress() {
            const wrapperRect = pinWrapper.getBoundingClientRect();
            const maxScroll = pinWrapper.offsetHeight - window.innerHeight;
            
            if (wrapperRect.top <= 0) {
                scrollProgress = Math.min(Math.max(-wrapperRect.top / maxScroll, 0), 1);
            } else {
                scrollProgress = 0;
            }
        }

        window.addEventListener('scroll', updateScrollProgress);

        // Mouse Parallax Interaction
        let mouseX = 0;
        let mouseY = 0;

        window.addEventListener('mousemove', (e) => {
            mouseX = (e.clientX - window.innerWidth / 2) * 0.0004;
            mouseY = (e.clientY - window.innerHeight / 2) * 0.0004;
        });

        // Window Resize
        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        });

        // --- Render & Animation Loop ---
        const clock = new THREE.Clock();

        function animate() {
            requestAnimationFrame(animate);
            const time = clock.getElapsedTime();

            // Continuous Subtle Idle Floating
            keyboardGroup.position.y = Math.sin(time * 1.2) * 0.4;

            // Mouse Parallax interpolation
            keyboardGroup.rotation.y += (mouseX - 0.3 - keyboardGroup.rotation.y) * 0.05;
            keyboardGroup.rotation.x += (mouseY + 0.55 - keyboardGroup.rotation.x) * 0.05;

            // --- Vertical Dismantling / Exploded View Sequence on Scroll ---
            
            // 1. Move Base Case downward
            caseMesh.position.y = -1 - (scrollProgress * 6);

            // 2. PCB Plate stays relatively grounded / slight shift
            pcbMesh.position.y = -scrollProgress * 2;

            // 3. Elevate Key Switches Vertically
            switchesGroup.position.y = scrollProgress * 5;

            // 4. Keycaps Lift & Stagger Vertically (Explode into air)
            keycapsList.forEach((cap) => {
                const stagger = cap.userData.delayFactor;
                const totalLift = (scrollProgress * 16) + (scrollProgress * stagger * 4);
                cap.position.y = cap.userData.initialY + totalLift;
            });

            // Rotate overall assembly slightly during scroll for better 3D depth perspective
            keyboardGroup.rotation.z = scrollProgress * 0.25;

            // Hero Text Fade Out as sequence completes
            if (scrollProgress > 0.35) {
                const fadeOpacity = 1 - ((scrollProgress - 0.35) / 0.5);
                heroContent.style.opacity = Math.max(fadeOpacity, 0);
                heroContent.style.transform = `scale(${1 - (scrollProgress - 0.35) * 0.15})`;
            } else {
                heroContent.style.opacity = 1;
                heroContent.style.transform = 'scale(1)';
            }

            renderer.render(scene, camera);
        }

        animate();
    </script>
